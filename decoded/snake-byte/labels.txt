# Snake Byte labels, for the `id` disassembler:
#
#     id
#     > loadb33 snake-byte.b33
#     > labels labels.txt
#     > dis $664A $66A8
#
# or non-interactively:
#
#     printf 'loadb33 snake-byte.b33\nlabels labels.txt\ndis $664A $66A8\n' | id
#
# Format: "ADDR name", hex, '$' optional, '#' starts a comment. These override
# the built-in Apple II symbol database, so game names win over ROM ones.
#
# Names are recorded here as they are established, with the evidence. Anything
# unproven belongs in a comment, not in a name.

# --- entry and startup -----------------------------------------------------
3750 relocate_1800     # copies $3800-$3FFF to $1800-$1FFF, then JMP $76C2
376E init_0300         # writes $52/$01/$01/$01/$64 to $0300-$0304
76C2 main              # target of the JMP at $3785

# --- the plotter's argument block -------------------------------------------
# $0000-$0008: nine bytes of zero page the drawing routines pass arguments in.
# $664A, the glyph blitter, reuses all nine for something unrelated -- it is a
# hand-written union, and the second set of names below is the other arm.
0000 var_shape         # which shape; hgr_set_masks turns it into the $6060 masks
0001 var_ink           # lo-res colour 0..15, 0 erases
0002 var_col           # the cell, on the 40x48 grid
0003 var_row
0004 var_hgr_dest      # 16-bit: the hi-res byte being written, +$400 per scanline
0006 var_dot_index     # entry in tbl_hgr_pattern
0007 var_scanline      # which of the cell's four, 0..3
0008 var_run_end       # last column of an hline / last row of a vline, inclusive
# The $664A arm, at the same addresses:
#   $0000 glyph source pointer into tbl_font
#   $0002 / $0003 the caller's X and Y, restored at $669F
#   $0008 the character, $20..$7F

6638 var_want_row

6B3A var_search_row

6C46 var_tone_period   # and the on/off switch: 0 is silent

6C71 var_joystick      # the player chose the joystick at the setup prompt

# --- the scoreboard --------------------------------------------------------
# $7252-$7266, all BCD, multi-byte fields least significant first. Six of them
# are what the status panel at $72CE prints, and its on-screen labels are the
# evidence for those six names.
7252 var_score          # SCORE, 4 bytes
7256 var_hi_score       # HI SCORE, 4 bytes
725A var_apples_left    # APPLES LEFT, 2 bytes -- the round's quota, counting down
725E var_lives          # SNAKES LEFT, 1 byte; $7691 sets it to 2
725F var_apples_afield  # 2 bytes: apples on the playfield now. $7642 increments
                        # it on placing one ($766C), $7743 decrements it on
                        # eating one, and $77D0 places a replacement exactly
                        # when it hits zero. Measured $01 at every apple eaten
                        # in play-hires, with $77D0 firing each time.
7261 var_apples_eaten   # 2 bytes, this round; only the first $11 score ($777B)
7263 var_apples_quota   # 2 bytes, copied into var_apples_left by $76C7
7265 var_level          # LEVEL, 1 BCD byte
7266 var_life_time      # what $6255 is loaded with when a life starts
71CB var_apple_value    # VALUE, 2 bytes; $71CD recomputes it per level
78B0 var_bonus_amount   # the bonus screen's 2-byte BCD scratch
78B2 var_lives_at_level # lives when the level began; $77F8 compares to decide
                        # whether the round earned a bonus
6255 var_life_timer     # counts down one per pace; $641C ends the life at zero.
                        # $6B55 also reads it as the side-wall height seed, and
                        # rewrites it to $FF whenever bit 7 is set -- which is
                        # what makes a $FF seed an untimed life rather than a
                        # long one.
0304 var_level_time     # the level's allowance, seeding var_life_time. Written
                        # by the display list's 'T' command. NOT the apple
                        # value: $71CD never reads it.

# --- the key-redefinition screen -------------------------------------------
# Reached by falling off the end of the inline string that hires_text_screen
# prints: the RTS at $7251 lands on $7541. Supplied to apple2tc via code-at.txt,
# since the recording never took that edge. The screen reads:
#
#         ABSOLUTE            RELATIVE
#        DIRECTION           DIRECTION
#            b                                 b/c/d/e are font glyphs
#          d + e               d   e           $E2/$E3/$E4/$E5 = the arrows
#            c
#       PRESS THE KEY FOR THIS FUNCTION
#
# Six slots: absolute up/left/right/down, then relative turn-left/turn-right.
7541 redefine_keys     # 6 slots: show current key, blink, read new, write back
6C63 tbl_keys          # 6 bytes, C9 CA CB CD 88 95 = I J K M left-arr right-arr
6C6A tbl_keys_default  # the same six bytes again, never written by $7541
7590 print_key_at      # A=char, X=slot; maps $88/$95 to glyphs $E6/$E7 for COUT
75B3 tbl_key_ch        # CH per slot for print_key_at: 0A 06 0E 0A 1A 22
75B9 tbl_key_cv        # CV per slot for print_key_at: 05 09 09 0D 09 09
75BF tbl_arrow_ch      # CH per slot for the blinking arrow: 0A 08 0C 0A 1C 20
75C5 tbl_arrow_cv      # CV per slot for the blinking arrow: 07 09 09 0B 09 09
75CB tbl_arrow_glyph   # E2 E4 E5 E3 E4 E5 = up left right down left right
75D1 read_key_for_slot # blinks the slot's arrow, returns an accepted key in A
761C accept_key        # >= $A1, or $88/$95; anything else resumes the blink

# Both $7590 and $75D1 store CH/CV then JSR $FC68, the tail of the ROM's LF
# (LDA CV / CMP WNDBTM / BCC VTABZ). Their CV is always well under WNDBTM, so
# it acts as a plain VTAB and never scrolls. See a2rom.h.

# --- the bonus display -----------------------------------------------------
# Same idiom: $7942 prints the inline string "BONUS: " and its RTS lands on
# $794D. Also supplied via code-at.txt -- $7942 is reached in the recording but
# the session never displayed a bonus, so the continuation was never traced.
793F show_bonus        # installs the hi-res hook, then prints through it
794D show_bonus_digits # two digits from $78B1/$78B0, then CSWL back to $FDF0

# --- the vector display list -----------------------------------------------
# $8000-$853D is *data*: 29 drawing scripts, one per level, in a byte stream.
# Not code, which is why nothing ever jumped into it. $7113 points $0A/$0B at
# $8000 and skips ($0303 - 1) '*'-terminated scripts to select one, then $7100
# interprets it. $7019 is the stream reader. Commands are ASCII:
#
#   'H' +4  colour, then HLINE (lo-res) and hgr_draw_hline (hi-res)
#   'V' +4  colour, then hgr_draw_vline
#   'P' +3  colour, then MON_PLOT and hgr_draw
#   'T' +1  stored to $0304
#   'E'     reset $0303 to 1 and restart -- the byte at $853D, ending the list
#   '*'     end of this script
#
# The grammar parses the whole region with no leftovers: 133 H, 129 V, 1 T,
# 29 '*', 1 'E'. Only H and V are actually used for level geometry.
7000 script_set_colour2 # helper used by the 'V' command
7019 script_next_byte  # LDA ($0A),Y / INC $0A -- fetch and advance
7024 script_set_colour # A = colour index, falls into SETCOL
7100 run_script        # interprets one display list
7113 select_script     # skips ($0303 - 1) scripts from $8000
8000 tbl_display_lists # 29 '*'-terminated drawing scripts
0303 var_script_index  # 1-based; init_0300 sets it to 1
0300 var_step_delay     # snake speed; $772E takes 2 off each life, floor 3
0301 var_difficulty     # 0..2; indexes the per-apple value table at $71C8
0302 var_demo_mode      # nobody answered the difficulty prompt, so the game
                        # plays itself. Any input clears it.
0305 var_head_moved     # the head moved this step, so the next draw merges the
                        # head shape over the cell. $6BEF raises, $6BDA clears.
000A var_script_ptr     # into the current display list
000C var_str_ptr        # into the string after a JSR to print_inline_str
000E var_rand_ptr       # $6C4B's cursor. Not a generator: it walks $1800-$1FFF
                        # and returns the first byte with bit 7 clear, so the
                        # numbers are the game's own level data, relocated
                        # there by $3750. $7980 clamps the high byte to
                        # [$18,$1F).
73D7 var_setup_seen     # the setup screen has run once; the first pass asks
                        # nothing and takes difficulty 1 plus demo mode

# --- video soft switches used once, at $704D-$7056 -------------------------
7045 set_video_mode    # $C050 graphics, $C057 hi-res, $C055 page 2, $C052 full
